* Git Profiles

** Main Configuration
#+begin_src shell :tangle ~/.gitconfig
  [user]
      name = randy1burrell
      email = me@randyburrell.info

  [core]
      editor = emacs
      excludesfile = /Users/randyburrell/.gitignore
      # autoSetupRemote = true

  [filter "lfs"]
      clean = git-lfs clean -- %f
      smudge = git-lfs smudge -- %f
      process = git-lfs filter-process
      required = true

  [github]
      user = randy1burrell
      email = me@randyburrell.info

  # Enforce SSH
  [url "ssh://git@github.com/"]
      insteadOf = https://github.com/

  [url "ssh://git@gitlab.com/"]
      insteadOf = https://gitlab.com/

  [url "ssh://git@bitbucket.org/"]
      insteadOf = https://bitbucket.org/

  [gpg]
      program = gpg

  [commit]
      gpgsign = true

  [alias]
      stag = "!id=`git rev-parse --verify HEAD`; git tag -s user_${id:0:8} -m \"Tag for commit $id\""
      vtag = !git tag -v `git describe`

  [includeIf "gitdir:~/"]
      path = ~/.gitconfig-personal

  [gitlab]
      user = randy1burrell
      email = randy1burrell@yahoo.com
  # Taken from https://emacsformacosx.com/tips
  [merge]
          tool = ediff
  [mergetool "ediff"]
          cmd = emacs --eval \"(ediff-merge-files-with-ancestor \\\"$LOCAL\\\" \\\"$REMOTE\\\" \\\"$BASE\\\" nil \\\"$MERGED\\\")\"
#+end_src
